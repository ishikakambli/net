using System; 
namespace p1even_odd 
{ 
class Program 
{ 
static void Main(string[] args) 
{ 
int num1 = 0; 
Console.WriteLine("Enter the number:"); 
num1 = Convert.ToInt32(Console.ReadLine()); 
if (num1 % 2 == 0) 
{ 
Console.WriteLine("The number is even."); 
} 
else 
{
Console.WriteLine("The number is odd."); 
} 
} 
} 
}
